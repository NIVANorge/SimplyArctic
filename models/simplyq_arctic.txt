
model("SimplyQ Arctic") {
	
	
    option_group("Hydrology structure") {
        include_qquick : par_bool("Include quick flow", false)
        compute_water_temp : par_bool("Compute water temperature", false)
    }
    
	sc     : index_set("Subcatchment")
	patch  : index_set("Patch")  # This is for both land cover and elevation

	air      : compartment("Atmosphere", sc, patch)
	snow_box : compartment("Snow box",   sc, patch)
	glacier  : compartment("Glacier",    sc, patch)
	soil     : compartment("Soil",       sc, patch)
	river    : compartment("River",      sc)
    subcatch : compartment("Subcatch",   sc)
	
	water       : quantity("Water")
	snow        : quantity("Snow (water equivalents)")
	ice         : quantity("Ice (water equivalents)")
	
	temp        : property("Temperature")
	precip      : property("Precipitation")
	flow        : property("Flow")
	pet         : property("Potential evapotranspiration")
    thaw_depth  : property("Soil thaw depth")
    melt        : property("Snow melt")
    snowfall    : property("Snowfall")
    liquid_rain : property("Rainfall")
    is_ice      : property("There is ice")
    is_snow     : property("There is snow")
    
    ice_epsilon : constant("Ice epsilon", [m m], 1e-6)     # Just for safe testing of ice thickness > 0
    t_melt      : constant("Melt temperature", [deg_c], 0) # Temperature at which snow and glacier ice start melting
	
	downhill   : connection("Downhill throughflow") @directed_graph { soil+ river } @no_cycles
	downstream : connection("Downstream") @directed_graph { river+ out } @no_cycles
	
	par_group("Catchment data", river) {
		a_catch    : par_real("Catchment area", [k m 2], 50, 1e-6, 7e6)
        elev_catch : par_real("Elevation of input met data", [m], 100, -420, 8848)
	}
	
	par_group("Patch data", soil) {
		rel_area : par_real("Relative area", [], 0.2, 0, 1)
	}
    
    # Unit conversions when going from patches to reaches
	unit_conversion(glacier.water, river.water) { a_catch }
	unit_conversion(soil.water, river.water)  { a_catch }
    unit_conversion(snow_box.water, river.water) { a_catch }
	
    # Aggregations
	aggregation_weight(soil, soil)     { rel_area / rel_area[below] }
	aggregation_weight(glacier, river) { rel_area }
	aggregation_weight(soil, river)    { rel_area }
    aggregation_weight(snow_box, river) { rel_area }
    
    # Aggregations for area-weighting rainfall, snow melt and glacial melt
    # across patches to get whole subcatchment averages
    aggregation_weight(snow_box, subcatch) { rel_area }
    aggregation_weight(glacier, subcatch) { rel_area }
    
    option(include_qquick) {
        downslope  : connection("Downhill quickflow") @directed_graph { snow_box+ river } @no_cycles
        
        aggregation_weight(snow_box, snow_box)     { rel_area / rel_area[below] }
        
        quick_target : loc(downslope)
    } @otherwise {
        quick_target : loc(out) # Just a dummy value that will not be used
    }
    
    # Setup solver
	simply_solver : solver("Simply solver", inca_dascru, [2, hr], 1e-3)
    
	solve(simply_solver, soil.water, river.water, glacier.water)
	solve(simply_solver, snow_box.water, snow_box.snow, glacier.ice)
    
    load("local_modules/atmosphere.txt", 
        module("ArcticAtmosphere", air, soil, temp, precip, elev_catch))
    
    load("local_modules/snow_glacier.txt",
        module("ArcticSnow", air, snow_box, subcatch, water, snow, temp, precip, flow, melt, snowfall, liquid_rain, is_snow, t_melt, ice_epsilon, loc(glacier.water), loc(glacier.ice), loc(glacier.is_ice), include_qquick, thaw_dpt, quick_target),
        module("SimplyGlacier", air, glacier, snow_box, subcatch, water, ice, snow, temp, flow, is_ice, t_melt, ice_epsilon, loc(soil.water), loc(river.water))
    )
        
    load("local_modules/soil.txt",
        module("SimplyPermafrostFlow", soil, water, flow, pet, loc(downhill)),
        module("Degree-day PET", air, soil, pet, temp, loc(glacier.is_ice), loc(snow_box.is_snow))
    )
        
    load("local_modules/river.txt",
        module("SimplyQ river manual", river, water, flow, a_catch, loc(downstream)))
        
    option(include_qquick) {
        # The quick flow formulation uses the soil thaw depth.
        
        thaw_dpt : loc(soil.thaw_depth)
        load("local_modules/soil.txt",
            module("SimplyPermafrost", air, soil, snow_box, glacier, snow, ice, temp, thaw_depth))
    } @otherwise {
        thaw_dpt : loc(constant(0)) #NOTE: This one is not used.
    }
    
    option(compute_water_temp) {
        
        # NOTE: Soil temperature is required for the water temperature module.
        # If it was already loaded above, Mobius2 knows that it should not load it twice.
        load("local_modules/soil.txt",
            module("SimplyPermafrost", air, soil, snow_box, glacier, snow, ice, temp, thaw_depth))
        
        heat : quantity("Heat energy")
        
        load("local_modules/water_temperature.txt", 
            module("Water temperature", air, glacier, snow_box, soil, river, water, heat, temp))
        
        # For averaging temperatures
        aggregation_weight(air, river) { rel_area }
    }
}
