module("ArcticSnow", version(0, 0, 0),
    air : compartment,
    snow_box : compartment,
    glacier : compartment,
    subcatch : compartment,
    water : quantity,
    snow : quantity,
    ice  : quantity,
    temp : property,
    precip : property,
    flow : property,
    t_melt : constant,
    ice_epsilon : constant,
    infiltr_target : loc,
    
    include_qquick : par_bool,
    thaw_depth : loc,
    quick_target : loc,
) {
"""
ODE-based version of the HBV-Nordic snow model.

Authors: Leah A. Jackson-Blake, Magnus D. Norling
"""

    load("stdlib/basic_math.txt", library("Response"))
	
    par_group("Snow") {
        ddf0_melt : par_real("Degree-day factor for snowmelt", [m m, deg_c-1, day-1], 2.74, 0, 6, "If DDF snowmelt amplitude is set to 0, this is a constant. If the amplitude is > 0, this is the maximum DDF, assumed to be in early summer")
        ddf_melt_dynamic: par_bool("Dynamic ddf_snow", true, "Calculate a dynamic sinusoidal ddf for snowmelt? See code for implementation details")
        refr_eff  : par_real("Refreeze efficiency", [], 0, 0, 1, "Speed of refreeze of liquid water in snow relative to speed of melt at the opposite temperature differential")
        snow_liq  : par_real("Liquid water fraction", [], 0, 0, 0.4, "How many mm liquid water one mm of snow (water equivalents) can hold before the water is released")
    }
    
    option(include_qquick) {
        par_group("Quick flow parameters") {
            max_infiltratn_fracn : par_real("Maximum infiltration fraction", [], 1, 0, 1, "Maximum proportion of rain and snowmelt that infiltrates to the soil box once 'ALD for maximum infiltration' is reached")
            ald_max_infiltration : par_real("ALD for maximum infiltration", [m m], 100, 0, 5000, "Active layer depth at which rain and snowmelt infiltration tend towards the maximum")
        }
    }

    par_group("Initial snow", snow_box) {
        init_snow : par_real("Initial snow depth (water equivalents)", [m m], 0, 0, 50000)
    }
    
    #assert("DDF snowmelt amplitude must be smaller than 0.5 * the Degree-day factor for snowmelt to #avoid negative DDF values", snow_box.snow) {
    #    ddf0_melt_amp < 0.5 * ddf0_melt
    #}

    # Constants
    t_snow : constant("Snow temp", [deg_c], 0) # Temperature below which precip falls as snow
    
    var(snow_box.snow,  [m m], "Snow depth") @initial { init_snow }
    var(snow_box.water, [m m], "Snow water")
    
    ddf : property("Degree-day factor")
    var(snow_box.ddf, [m m, deg_c-1, day-1], "Snow melt degree-day factor") {
        ddf0_melt if !ddf_melt_dynamic,
        {
            ddf_mean := ddf0_melt/2,
            ddf_mean + ddf_mean*cos(2*pi*(time.day_of_year=>[] - 165) / 365)
        } otherwise
    }
    
    
    # FLUXES
    
    snowfall : property("Snowfall")
    liquid_rain : property("Rainfall")
    melt : property("Snow melt")
    
    var(snow_box.snowfall, [m m, day-1], "Precipitation falling as snow") { air.precip * (air.temp <= t_snow)  } @no_store

    flux(out, snow_box.snow, [m m, day-1], "Precipitation falling as snow") { snowfall }

    var(snow_box.liquid_rain, [m m, day-1], "Rainfall") { air.precip * (air.temp > t_snow) } @no_store

    flux(out, snow_box.water, [m m, day-1], "Precipitation as rain")  { liquid_rain }

    # Need a separate property if want to aggregate over the patches,
    # as can't access the values of a flux directly
    var(snow_box.snow.melt, [m m, day-1], "Snow melt") {

        pot_melt := max(0, ddf*(air.temp - t_melt)),
        s_response(snow, 0, 1[m m], 0, pot_melt)
    } @no_store
    
    flux(snow_box.snow, snow_box.water, [m m, day-1], "Snow melt flux") { melt }

    flux(snow_box.water, snow_box.snow, [m m, day-1], "Refreeze") {
        pot_refr := max(0, refr_eff * ddf0_melt * (t_melt - air.temp)),
        s_response(water, 0, 1[m m], 0, pot_refr)
    }

    # Tried, but didn't help
    #water_holding_capacity : property("Water holding capacity")
    #var(snow_box.snow.water_holding_capacity, [], "Snow water holding capacity") {
    #    s_response(snow_box.snow, 100[m m], 300[m m], 0, snow_liq)
    #}
    
    var(snow_box.water.flow, [m m, day-1], "Snow box outflow") {
        # The tc just needs to be low enough to be effectively instant (but not so low that it causes numerical instability)
        tc := 0.1[day],
        max(0, water - snow*snow_liq) / tc
        #max(0, water - snow * snow_box.snow.water_holding_capacity) / tc   otherwise
    } @no_store

    option(include_qquick) {
        
        infiltration : property("Infiltration")
        var(snow_box.water.infiltration, [m m, day-1]) { 
            # Assume a simple sigmoid relationship between infiltration and thaw depth in non-glaciated areas.
            # Assume 0 infiltration when thaw depth=0, all flow goes through soil when thaw depth = z.
            
            flow if glacier.ice > ice_epsilon,  # All goes to glacier.water if have glacier ice
            s_response(thaw_depth, 0[m m], ald_max_infiltration, 0, max_infiltratn_fracn * flow) otherwise
        }
        
        flux(snow_box.water, infiltr_target, [m m, day-1], "Precip & melt infiltration") { infiltration }
        
        flux(snow_box.water, quick_target, [m m, day-1], "Quick flow") { flow - infiltration }
    
    } @otherwise {

        flux(snow_box.water, infiltr_target, [m m, day-1], "Precip & melt infiltration") {  flow  }
    }
    
    # Aggregate over sub-catchment for convenient display
    var(subcatch.melt, [m m, day-1], "Subcatchment snow melt") {
        aggregate(snow_box.snow.melt)
    }
    
    var(subcatch.liquid_rain, [m m, day-1], "Subcatchment rainfall") {
        aggregate(snow_box.liquid_rain)
    }
    
    var(subcatch.snowfall, [m m, day-1], "Subcatchment snowfall") {
        aggregate(snow_box.snowfall)
    }
}

module("SimplyGlacier", version(0, 0, 1),
    air : compartment,
    glacier : compartment,
    snow_box : compartment,
    subcatch : compartment,
    water : quantity,
    ice : quantity,
    snow : quantity,
    temp : property,
    flow : property,
    t_melt : constant,
    ice_epsilon : constant,
    infiltr_target : loc,
    runoff_target : loc,
) {
"""
Simple glacier melt model.
"""
        
	load("stdlib/basic_math.txt", library("Response"))
	
	par_group("Glacier global") {
		ddf_ice : par_real("Degree-day factor for ice melt", [m m, deg_c-1, day-1], 3, 1, 15)
		#snowtoice : par_real("Glacier accumulation coefficient", [day-1], 0, 0, 1e-3, "Proportion of overlying snow which becomes glacier ice (for roughly appropriate glacier mass balance)")
		mpf: par_real("Glacier melt percolation fraction", [], 0, 0, 1, "Proportion of glacial melt that percolates down to the soil (vs going to the reach)")
	}
	
	par_group("Glacier", glacier) {
		init_ice : par_real("Initial ice thickness", [m m], 0)
	}
	
	# Variables
	var(glacier.ice, [m m], "Glacier ice") @initial { init_ice }
	
	var(glacier.water, [m m], "Glacier compartment water")
	
	var(glacier.water.flow, [m m, day-1], "Glacier compartment inflow") {
		#in_flux(water)
        tc := 0.1[day],  # 0.01 instead of 0.05 doubles the model run time
            # Also add a "fake" field capacity for numerical reasons
            # TODO: where have glaciers, probably want a big field capacity
        max(0, (water - 1[m m]) / tc)
	}
    # Seibert et al. (2018) have a more complex version of this, varying with snow
    # depth. However, don't have glacier melt unless snow is gone, so simplify for now
    #var(glacier.water.flow, [m m, day-1], "Glacier compartment flow") {
        #water * (K_min + K_range * exp(-A_G * aggregate(snow_box.snow)))            
    #}
	
	# FLUXES
	# When glacier ice is not present in this compartment (i.e. when the initial glacier ice thickness for this patch = 0), then this compartment is entirely passive. Fluxes from the snow compartment just pass straight through to the soil water compartment and no glacier accumulation occurs. When glacier ice is present, accumulation and melt occur, and melt water either goes directly to the river reach or percolates to the soil water compartment.

	# flux(snow_box.snow, glacier.ice, [m m, day-1], "Glacier accumulation") {
	# # NOTE this needs revisiting. At the moment, the higher snowtoice is, the MORE glacier ice is lost over the model run (because then the snow disappears quicker and ice melt starts sooner, and ddf for ice melt tends to be higher than for snow melt, which the model seems to like).
		# snowtoice * snow   if last(glacier.ice) > ice_epsilon,
		# 0                  otherwise
	# }
	
	glac_melt : property("Glacier melt")
	var(glacier.ice.glac_melt, [m m, day-1]) {
		# Smoothly turn-off glacial melt as a function of snow depth (melt goes from max_melt to 0 as snow depth goes from 0 to 5 cm. Quick literature review suggested 5 cm is about appropriate)
		max_melt := min(last(glacier.ice)*1[day-1], max(0, ddf_ice*(air.temp - t_melt))),
		s_response(aggregate(snow_box.snow), 0[m m], 50[m m], max_melt, 0[m m])
	} @no_store
    #TODO: What is the aggregate() around snow_box.snow for? glacier and snow_box are distributed the same.

	flux(glacier.ice, glacier.water, [m m, day-1], "Glacier melt flux") { glac_melt }
	
	flux(glacier.water, runoff_target, [m m, day-1], "Glacial runoff") {
	# Flux of water going directly to the reach (e.g. supraglacial or subglacial streams)
	# In this simple setup, it's all the flow if have ice, none of it if don't have ice.
		flow*(1 - mpf)  if ice > ice_epsilon,
		0               otherwise
	}
	
	flux(glacier.water, infiltr_target, [m m, day-1], "Input to soil water") {
	# Flux of glacial water down to the soil. For river fluxes (not caring so much about flow pathways), set this extremely simply to just be 0 when have glacial ice, i.e. all meltwater goes straight to supraglacial or subglacial channels. However, this is a huge simplification. TODO: have a proportion of meltwater that goes to the soil. This could be constant, or could vary e.g. depending on whether the glacier is warm or cool-based, time of year, etc. Probably need geochem data to parameterise this.
		flow*mpf     if ice > ice_epsilon,
		flow         otherwise
	}
	
	# Aggregate melt over sub-catchment for convenient display 
	var(subcatch.glac_melt, [m m, day-1], "Subcatchment glacial melt") {
		aggregate(glacier.ice.glac_melt)
	}

}
